{{- if .Values.dashboard.allowWrites }}

  !! scans -- and register artifacts, if allowManualRegistration is on
  !! too. Reasonable on a private network; make sure that is what this

{{- if not .Values.monitorApi.apiKeyScopes }}

  !! SCOPES ARE NOT ENFORCED

  !! monitorApi.apiKeyScopes is empty, so every key -- named or not --
